# === Load and preprocess dataset ===
data = pd.read_csv("/home/techub/NF-ToN-IoT-Sample.csv")
X = data.drop(['IPV4_SRC_ADDR', 'IPV4_DST_ADDR', 'Attack', 'Label'], axis=1)
y = data['Label']
X, y = shuffle(X, y, random_state=42)
feature_index = {feature: idx for idx, feature in enumerate(X.columns)}
feature_names = list(feature_index.keys())

# === Define and train model ===
base_model = DeepOnlineClassifierWithDynamicQuantization(
    feature_index=feature_index,
    input_dim=len(feature_index),
    hidden_dim=64,
    output_dim=32,
    deep_lr=0.001,
    activation='leaky_relu',
    dropout=0.2,
    seed=42
)

model = compose.Pipeline(
    preprocessing.StandardScaler(),
    base_model
)

# Train on clean stream
for i, (x_i, y_i) in enumerate(stream.iter_pandas(X, y)):
    model.learn_one(x_i, y_i)
    if i > 5000:
        break

# === Inference on clean inputs ===
clean_preds = []
true_labels = []
X_list = []

for i, (x_i, y_i) in enumerate(stream.iter_pandas(X, y)):
    y_pred = model.predict_one(x_i)
    clean_preds.append(y_pred)
    true_labels.append(y_i)
    X_list.append(x_i)
    if i >= 20000:
        break

# === Apply quantization ===
base_model.quantize_model()

# === Evaluate prediction under noise and adversarial perturbation ===

def add_gaussian_noise(x: dict[str, float], std_dev: float = 0.01) -> dict[str, float]:
    return {k: v + np.random.normal(0, std_dev) for k, v in x.items()}

def add_adversarial_shift(x: dict[str, float], eps: float = 0.05) -> dict[str, float]:
    return {k: v + eps if i % 2 == 0 else v - eps for i, (k, v) in enumerate(x.items())}

noise_preds = []
adv_preds = []

for x_clean in X_list:
    x_noisy = add_gaussian_noise(x_clean, std_dev=0.02)
    x_adv = add_adversarial_shift(x_clean, eps=0.03)

    y_pred_noise = model.predict_one(x_noisy)
    y_pred_adv = model.predict_one(x_adv)

    noise_preds.append(y_pred_noise)
    adv_preds.append(y_pred_adv)

# === Robustness Scores ===
def consistency_score(base_preds, perturbed_preds):
    return np.mean(np.array(base_preds) == np.array(perturbed_preds))

robust_noise = consistency_score(clean_preds, noise_preds)
robust_adv = consistency_score(clean_preds, adv_preds)

print("\n--- Robustness Analysis ---")
print(f"Prediction Consistency w/ Gaussian Noise: {robust_noise:.4f}")
print(f"Prediction Consistency w/ Adversarial Perturbation: {robust_adv:.4f}")

# === Accuracy under perturbed inputs ===
acc_noise = accuracy_score(true_labels, noise_preds)
acc_adv = accuracy_score(true_labels, adv_preds)
acc_clean = accuracy_score(true_labels, clean_preds)

print(f"\nAccuracy on Clean Data: {acc_clean:.4f}")
print(f"Accuracy on Noisy Data: {acc_noise:.4f}")
print(f"Accuracy on Adversarial Data: {acc_adv:.4f}")

import pandas as pd
import numpy as np
from sklearn.utils import shuffle
from river import stream, preprocessing, compose, metrics
import matplotlib.pyplot as plt

# === Load and preprocess dataset ===
data = pd.read_csv("/home/techub/NF-ToN-IoT-Sample.csv")

# Shuffle entire dataset
data = shuffle(data, random_state=42)

# Simulate concept drift by splitting into two distributions
drift_point = len(data) // 4

# Pre-drift: mostly normal traffic
pre_drift = data.iloc[:drift_point].copy()
pre_drift = shuffle(pd.concat([
    pre_drift[pre_drift['Label'] == 0],
    pre_drift[pre_drift['Label'] == 1].sample(frac=0.2, random_state=42)
]))

# Post-drift: increase attack ratio
post_drift = data.iloc[drift_point:].copy()
post_drift = shuffle(pd.concat([
    post_drift[post_drift['Label'] == 1],
    post_drift[post_drift['Label'] == 0].sample(frac=0.2, random_state=42)
]))

# Concatenate to simulate a stream with concept drift
data_stream = pd.concat([pre_drift, post_drift]).reset_index(drop=True)

# Extract features and labels
X = data_stream.drop(['IPV4_SRC_ADDR', 'IPV4_DST_ADDR', 'Attack', 'Label'], axis=1)
y = data_stream['Label']
feature_index = {feature: idx for idx, feature in enumerate(X.columns)}

# === Define pipeline ===
model = compose.Pipeline(
    preprocessing.StandardScaler(),
    DeepOnlineClassifierWithDynamicQuantization(
        feature_index=feature_index,
        input_dim=len(feature_index),
        hidden_dim=64,
        output_dim=32,
        deep_lr=0.001,
        activation='leaky_relu',
        dropout=0.2,
        seed=42
    )
)

# === Track performance over time ===
window_size = 100
accuracies = []
sliding_window = []

# Initialize accuracy metric
acc = metrics.Accuracy()

# Simulate online learning and concept drift adaptation
for i, (x_i, y_i) in enumerate(stream.iter_pandas(X, y)):
    y_pred = model.predict_one(x_i)
    model.learn_one(x_i, y_i)

    # Update metric
    acc.update(y_i, y_pred)

    # Sliding window accuracy recording
    sliding_window.append(int(y_pred == y_i))
    if len(sliding_window) > window_size:
        sliding_window.pop(0)
    if i % 1000 == 0:
        accuracy = sum(sliding_window) / len(sliding_window)
        accuracies.append((i, accuracy))

# === Plot accuracy over time to visualize drift impact ===
indices, acc_vals = zip(*accuracies)
plt.figure(figsize=(10, 5))
plt.axvline(x=drift_point, color='r', linestyle='--', label='Concept Drift Point')
plt.plot(indices, acc_vals, label='Sliding Window Accuracy')
plt.xlabel('Stream Index')
plt.ylabel('Accuracy')
plt.title('Online Adaptation to Concept Drift')
plt.legend()
plt.grid(True)
plt.tight_layout()
plt.savefig('Label_Drift_ToN.png')
plt.show()

import pandas as pd

from sklearn.utils import shuffle

# Load the dataset
data = pd.read_csv("/home/techub/NF-ToN-IoT-Sample.csv")

# Split into features and target

y = data['Label']

# Shuffle the data
X, y = shuffle(X, y, random_state=42)

# Create feature index
feature_index = {feature: idx for idx, feature in enumerate(X.columns)}


# Helper function to create a fresh model
def create_pipeline():
    return compose.Pipeline(
        preprocessing.StandardScaler() |
        DeepOnlineClassifierWithDynamicQuantization(
            feature_index=feature_index,
            input_dim=len(feature_index),
            hidden_dim=64,
            output_dim=32,
            deep_lr=0.001,
            activation='leaky_relu',
            dropout=0.2,
            seed=42
        )
    )

# Define different packet arrival rates to simulate
stream_rates = [100, 500, 1000, 5000, 10000]  # samples per second

# Results dictionary
results = {}

# Conduct test for each stream rate
for rate in stream_rates:
    print(f"\n=== Testing at {rate} samples/sec ===")

    # Create a fresh pipeline for each run
    classifier = create_pipeline()

    # Initialize counters
    sample_count = 0
    total_latency = 0.0
    start_time = time.perf_counter()

    for xi, yi in stream.iter_pandas(X, y):
        t0 = time.perf_counter()

        # Predict
        y_pred = classifier.predict_one(xi)
        if y_pred is None:
            y_pred = 0

        # Learn
        classifier.learn_one(xi, yi)

        t1 = time.perf_counter()

        # Measure latency
        latency = (t1 - t0)
        total_latency += latency
        sample_count += 1

        # Simulate incoming packet rate
        expected_interval = 1.0 / rate
        elapsed = t1 - t0
        sleep_time = max(0.0, expected_interval - elapsed)
        time.sleep(sleep_time)

        # Limit number of samples processed (e.g., 3000 samples)
        if sample_count >= 10000:
            break

    end_time = time.perf_counter()
    total_time = end_time - start_time

    actual_pps = sample_count / total_time
    avg_latency_ms = (total_latency / sample_count) * 1000  # Convert to milliseconds

    results[rate] = {
        'Samples': sample_count,
        'Total Time (s)': total_time,
        'Actual PPS': actual_pps,
        'Avg Inference+Learning Latency (ms)': avg_latency_ms,
    }

# Print Final Results

    print(f"Samples Processed: {res['Samples']}")

import pandas as pd
from collections import Counter
from sklearn.utils import shuffle
from river import stream, preprocessing, compose, metrics
import matplotlib.pyplot as plt

# Load the dataset
data = pd.read_csv("/home/techub/NF-ToN-IoT-Sample.csv")

# Split into features and target
X = data.drop(['IPV4_SRC_ADDR', 'IPV4_DST_ADDR', 'Attack', 'Label'], axis=1)
y = data['Label']

# Shuffle the data
X, y = shuffle(X, y, random_state=42)

feature_index = {feature: idx for idx, feature in enumerate(X.columns)}

# === Define classifier pipeline ===
classifiers = {
    'DynamicQuantizedDeepOnlineClassifier': compose.Pipeline(
        preprocessing.StandardScaler(),
        DeepOnlineClassifierWithDynamicQuantization(
            feature_index=feature_index,
            input_dim=len(feature_index),
            hidden_dim=64,
            output_dim=32,
            deep_lr=0.001,
            activation='leaky_relu',
            dropout=0.2,
            seed=42
        )
    )
}

# === Initialize Metrics ===
metrics_dict = {
    name: metrics.LogLoss()
    for name in classifiers
}

# Track LogLoss over time
logloss_history = {name: [] for name in classifiers}

# === Online Training Loop ===
for i, (x, true_y) in enumerate(stream.iter_pandas(X, y)):
    for name, model in classifiers.items():
        y_pred = model.predict_proba_one(x)
        model.learn_one(x, true_y)
        metrics_dict[name].update(true_y, y_pred)

        # Record current LogLoss
        logloss_history[name].append(metrics_dict[name].get())

# === Plot LogLoss ===
plt.figure(figsize=(10, 6))
for name, losses in logloss_history.items():
    plt.plot(losses, label=name)

plt.title("LogLoss over Time")
